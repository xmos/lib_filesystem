cmake_minimum_required(VERSION 3.21)
include($ENV{XMOS_CMAKE_PATH}/xcommon.cmake)
project(test_service)
set(XMOS_SANDBOX_DIR ${CMAKE_CURRENT_LIST_DIR}/../../..)
set(APP_HW_TARGET src/XK-EVK-XU316-AIV.xn CACHE STRING "Test hardware target")
set(APP_INCLUDES src)
set(FS_TEST_CONFIG_DIR "" CACHE PATH "Application configuration header directory (default: multi-client FatFs fixture)")
if(NOT FS_TEST_CONFIG_DIR)
    set(FS_TEST_CONFIG_DIR "${CMAKE_CURRENT_LIST_DIR}/../config/fatfs")
endif()
list(APPEND APP_INCLUDES "${FS_TEST_CONFIG_DIR}")
set(FS_TEST_IMAGE "" CACHE FILEPATH "128-sector FAT12 image populated by the native test")
if(NOT EXISTS "${FS_TEST_IMAGE}")
    message(FATAL_ERROR "Pass FS_TEST_IMAGE: create a 128-sector image containing TEST0.BIN, TEST1.BIN and EMPTY.BIN with fat_image_creator")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/../image_to_header.py
    "${FS_TEST_IMAGE}" "${CMAKE_CURRENT_BINARY_DIR}/generated/image.h" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${FS_TEST_IMAGE}" "${CMAKE_CURRENT_LIST_DIR}/../image_to_header.py")
list(APPEND APP_INCLUDES ${CMAKE_CURRENT_BINARY_DIR}/generated)
set(APP_COMPILER_FLAGS -O2 -g -report -fxscope)
set(FS_TEST_MEDIA block CACHE STRING "Test media connection: block, bridge or legacy")
set_property(CACHE FS_TEST_MEDIA PROPERTY STRINGS block bridge legacy)
if(FS_TEST_MEDIA STREQUAL "bridge")
    list(APPEND APP_COMPILER_FLAGS -DTEST_BRIDGE=1)
elseif(FS_TEST_MEDIA STREQUAL "legacy")
    list(APPEND APP_COMPILER_FLAGS -DTEST_LEGACY=1)
elseif(NOT FS_TEST_MEDIA STREQUAL "block")
    message(FATAL_ERROR "Unknown FS_TEST_MEDIA")
endif()
option(FS_TEST_BASIC "Exercise the legacy read-only file interface" OFF)
if(FS_TEST_BASIC)
    list(APPEND APP_COMPILER_FLAGS -DTEST_BASIC=1)
endif()
include(${CMAKE_CURRENT_LIST_DIR}/../../deps.cmake)
set(APP_XSCOPE_SRCS config.xscope)
XMOS_REGISTER_APP()
set_target_properties(${APP_BUILD_TARGETS} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
